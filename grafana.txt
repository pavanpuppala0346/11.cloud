        q.domain,
        q.subdomain,
        q.question_type,
        gt.relevancy_rating,
        gt.accuracy_rating,
        gt.completeness_rating,
        COALESCE(
            gt.overall_score,
            (gt.relevancy_rating + gt.accuracy_rating + gt.completeness_rating)::numeric / 3.0
        ) AS final_score
    FROM public.questions q
    LEFT JOIN public.ground_truth gt
      ON gt.question_id = q.id   -- adjust if actual FK differs
)
SELECT 
    COUNT(DISTINCT q.session_id) AS total_questions,

    AVG(final_score)::numeric            AS avg_final_score
FROM public.questions q

  ON gt.question_id = q.id;

WITH extracted AS (
    SELECT 
        (evaluation_scores->>'component_a')::numeric AS comp_a,
        (evaluation_scores->>'component_b')::numeric AS comp_b,
        (evaluation_scores->>'component_c')::numeric AS comp_c,
        (evaluation_scores->>'component_d')::numeric AS comp_d,
        (evaluation_scores->>'aggregate')::numeric   AS agg
    FROM public.generated_content
    WHERE evaluation_scores IS NOT NULL
)
SELECT 'Component A: Quality Metrics' AS component, AVG(comp_a) AS avg_score FROM extracted
UNION ALL
SELECT 'Component B: Automated Metrics', AVG(comp_b) FROM extracted
UNION ALL
SELECT 'Component C: LLM-as-a-Judge', AVG(comp_c) FROM extracted
UNION ALL
SELECT 'Component D: Human Feedback', AVG(comp_d) FROM extracted
UNION ALL
SELECT 'Aggregate Score', AVG(agg) FROM extracted;
